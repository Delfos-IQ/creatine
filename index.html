    function cambiarPestana(tab) {

        document.querySelectorAll('.app-section').forEach(s => s.classList.remove('active'));

        document.getElementById(`sec-${tab}`).classList.add('active');

    }

        let total = diccionario[idiomaActual].habs.length;

        let marcados = 0;

        for(let i=0; i<total; i++) {

            if(document.getElementById(`chk-${i}`).checked) marcados++;

        }

        let porcentajeHoy = Math.round((marcados / total) * 100);

        let historial = JSON.parse(localStorage.getItem('mtn_historial')) || [];

        historial.push({ fecha: new Date().toISOString().slice(0,10), porc: porcentajeHoy });

        localStorage.setItem('mtn_historial', JSON.stringify(historial));

        alert(idiomaActual === 'es' ? '¡Progreso guardado!' : 'Progresso guardado!');

        cargarHistorial();

        volverHome();

    }


    function cargarHistorial() {

        let historial = JSON.parse(localStorage.getItem('mtn_historial')) || [];

        if(historial.length === 0) {

            document.getElementById('stats-display').innerText = "0%";

            return;

        }

        let suma = historial.reduce((acc, curr) => acc + curr.porc, 0);

        let media = Math.round(suma / historial.length);

        document.getElementById('stats-display').innerText = `${media}%`;

    }


    function borrarHistorial() {

        localStorage.removeItem('mtn_historial');

        document.getElementById('stats-display').innerText = "0%";

    }


    function toggleTimer() {

        const btn = document.getElementById('btn-timer-start');

        if (timerRunning) {

            clearInterval(timerInterval);

            btn.innerText = "Play";

            btn.style.background = "var(--success)";

        } else {

            timerInterval = setInterval(() => {

                if(timerSeconds <= 0) {

                    clearInterval(timerInterval);

                    alert("¡Entrenamiento Completado!");

                    resetTimer();

                    return;

                }

                timerSeconds--;

                let mins = Math.floor(timerSeconds / 60);

                let secs = timerSeconds % 60;

                document.getElementById('display').innerText = `${mins.toString().padStart(2,'0')}:${secs.toString().padStart(2,'0')}`;

                if (secs === 0 && timerSeconds > 0) {

                    document.body.style.background = "#ecc94b";

                    setTimeout(() => document.body.style.background = "var(--bg-color)", 500);

                }

            }, 1000);

            btn.innerText = "Pause";

            btn.style.background = "#ecc94b";

        }

        timerRunning = !timerRunning;

    }


    function resetTimer() {

        clearInterval(timerInterval);

        timerSeconds = 1200;

        timerRunning = false;

        document.getElementById('display').innerText = "20:00";

        document.getElementById('btn-timer-start').innerText = "Play";

        document.getElementById('btn-timer-start').style.background = "var(--success)";

    }
